package client

import (
	"context"
	"errors"
	"io"
	"net/http"
	"strings"
	"testing"
	"time"
)

func TestNeutralSuppliedClientAlreadyErasesReadCause(t *testing.T){
	cause:=errors.New("independent stream cause")
	closes:=0
	c:=&http.Client{Timeout:15*time.Millisecond,Transport:reviewRT(func(r *http.Request)(*http.Response,error){
		return reviewResponse(200,&reviewCloseBody{reader:reviewReader(func([]byte)(int,error){<-r.Cancel;return 0,errors.Join(cause,r.Context().Err())}),closes:&closes}),nil
	})}
	req,_:=http.NewRequestWithContext(context.Background(),"GET","http://example.invalid",nil)
	resp,err:=c.Do(req)
	if err!=nil {t.Fatal(err)}
	_,err=io.ReadAll(resp.Body)
	resp.Body.Close()
	t.Logf("direct client cause=%t deadline=%t closes=%d error=%v",errors.Is(err,cause),errors.Is(err,context.DeadlineExceeded),closes,err)
	if errors.Is(err,cause) || closes!=1 {t.Error("control did not reproduce SDK erasure")}
}

func TestNeutralPolicyStatusBodyOwnership(t *testing.T){
	cause:=errors.New("redirect denied")
	closes:=0
	calls:=0
	c:=&http.Client{Transport:reviewRT(func(*http.Request)(*http.Response,error){
		calls++
		resp:=reviewResponse(302,&reviewCloseBody{reader:strings.NewReader("redirect"),closes:&closes})
		resp.Header.Set("Location","/next")
		return resp,nil
	}),CheckRedirect:func(*http.Request,[]*http.Request)error{return cause}}
	_,err:=Fetch(context.Background(),c,"http://example.invalid/start")
	t.Logf("status=%t cause=%t calls=%d closes=%d",strings.Contains(err.Error(),"HTTP 302"),errors.Is(err,cause),calls,closes)
	if !strings.Contains(err.Error(),"HTTP 302") || !errors.Is(err,cause) || calls!=1 || closes!=1 {t.Error("known status or ownership lost")}
}
