package signedcounter_test
import("bytes";"crypto/hmac";"crypto/sha256";"sync";"testing";sc "example.com/signedcounter")
func TestReviewerKeyOwnershipAndConstructorBoundary(t *testing.T){for _,n:=range []int{0,15}{if c,e:=sc.New(make([]byte,n));e==nil||c!=nil{t.Fatal("invalid key accepted",n)}};key:=[]byte("sixteen-byte-key!");owned:=bytes.Clone(key);c,e:=sc.New(key);if e!=nil||c.Count()!=0{t.Fatal(e)};key[0]^=255;data:=[]byte("data");mac:=hmac.New(sha256.New,owned);mac.Write(data);want:=mac.Sum(nil);first:=c.Sign(data);if !bytes.Equal(first,want)||string(data)!="data"{t.Fatal("key not owned or input mutated")};first[0]^=255;second:=c.Sign(data);if !bytes.Equal(second,want)||c.Count()!=2{t.Fatal("signature/state shared",c.Count())}}
func TestReviewerCountMonotonicUnderConcurrentSigning(t *testing.T){c,e:=sc.New([]byte("sixteen-byte-key!"));if e!=nil{t.Fatal(e)};const workers=8;const each=128;start:=make(chan struct{});done:=make(chan struct{});var wg sync.WaitGroup;for i:=0;i<workers;i++{wg.Add(1);go func(){defer wg.Done();<-start;for j:=0;j<each;j++{c.Sign(nil);_=c.Count()}}()};go func(){wg.Wait();close(done)}();close(start);var last uint64;for {got:=c.Count();if got<last||got>workers*each{t.Fatal("invalid count",last,got)};last=got;select{case <-done:if c.Count()!=workers*each{t.Fatal(c.Count())};return;default:}}}
