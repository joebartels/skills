package quotas_test

import (
	"testing"

	quotas "example.com/quotas"
)

func TestControlPositiveValuesOnly(t *testing.T) {
	for _, tc := range []struct {
		name   string
		values []int
		want   int
	}{
		{"nil", nil, 0},
		{"empty", []int{}, 0},
		{"nonpositive", []int{-5, 0, -1}, 0},
		{"positive", []int{2, 3}, 5},
		{"mixed", []int{-4, 3, 0, -2, 8}, 11},
	} {
		t.Run(tc.name, func(t *testing.T) {
			if got := quotas.Total(tc.values); got != tc.want {
				t.Errorf("Total = %d, want %d", got, tc.want)
			}
		})
	}
}
