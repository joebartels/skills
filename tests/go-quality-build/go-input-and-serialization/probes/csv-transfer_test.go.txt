package catalogue_test

import (
	"errors"
	"io"
	"reflect"
	"strconv"
	"strings"
	"testing"

	"example.com/catalogue"
)

var _ func(io.Reader, func([]catalogue.Item) error) error = catalogue.Import

func transferStock(n uint64) *uint64 { return &n }

func transferImportItems(t *testing.T, input string, want []catalogue.Item) {
	t.Helper()
	calls := 0
	err := catalogue.Import(strings.NewReader(input), func(items []catalogue.Item) error {
		calls++
		if len(items) != len(want) || (len(want) > 0 && !reflect.DeepEqual(items, want)) {
			t.Errorf("replacement = %#v, want %#v", items, want)
		}
		return nil
	})
	if err != nil || calls != 1 {
		t.Fatalf("Import error = %v, replacement calls = %d; want success and one call", err, calls)
	}
}

func transferRejectImport(t *testing.T, input string) {
	t.Helper()
	calls := 0
	err := catalogue.Import(strings.NewReader(input), func([]catalogue.Item) error {
		calls++
		return nil
	})
	if err == nil || calls != 0 {
		t.Fatalf("Import error = %v, replacement calls = %d; want error before replacement", err, calls)
	}
}

func TestTransferCSVCompleteCatalogue(t *testing.T) {
	transferImportItems(t, "id,name,stock\n9007199254740993,\"Upper\nshelf\",\n18446744073709551615,\"Bolt, large\",0\n", []catalogue.Item{
		{ID: 9007199254740993, Name: "Upper\nshelf"},
		{ID: 18446744073709551615, Name: "Bolt, large", Stock: transferStock(0)},
	})
	transferImportItems(t, "\xef\xbb\xbfid,name,stock\r\n7,\"The \"\"special\"\"\r\npart\",18446744073709551615\r\n8,  intact  ,\r\n", []catalogue.Item{
		{ID: 7, Name: "The \"special\"\npart", Stock: transferStock(18446744073709551615)},
		{ID: 8, Name: "  intact  "},
	})
	transferImportItems(t, "id,name,stock\n\n", nil)
}

func TestTransferCSVHeader(t *testing.T) {
	for _, header := range []string{
		"code,name,stock", "name,id,stock", "id,name", "id,name,stock,extra",
		"id,name,stock ", "\xef\xbb\xbf\xef\xbb\xbfid,name,stock", "id,\xef\xbb\xbfname,stock",
	} {
		t.Run(header, func(t *testing.T) { transferRejectImport(t, header+"\n1,Bolt,2\n") })
	}
	transferRejectImport(t, "")
	transferRejectImport(t, "\n\n")
}

func TestTransferCSVLateFailureIsAtomic(t *testing.T) {
	for name, tail := range map[string]string{
		"unterminated quote": "2,\"never closed,3\n",
		"bare quote":         "2,broken\"name,3\n",
		"too few fields":     "2,Washer\n",
		"extra field":        "2,Washer,3,extra\n",
		"duplicate ID":       "1,Another name,4\n",
		"zero ID":            "0,Washer,3\n",
		"empty name":         "2,,3\n",
		"blank name":         "2,  \t ,3\n",
		"whitespace record":  "  \n",
	} {
		t.Run(name, func(t *testing.T) { transferRejectImport(t, "id,name,stock\n1,Bolt,2\n"+tail) })
	}
}

func TestTransferCSVIntegerGrammar(t *testing.T) {
	for _, value := range []string{"+2", "-2", " 2", "2 ", "2.0", "2e3", "18446744073709551616", "\xef\xbb\xbf2"} {
		t.Run("id/"+value, func(t *testing.T) { transferRejectImport(t, "id,name,stock\n"+value+",Bolt,0\n") })
		t.Run("stock/"+value, func(t *testing.T) { transferRejectImport(t, "id,name,stock\n1,Bolt,"+value+"\n") })
	}
	transferImportItems(t, "id,name,stock\n0001,Bolt,0000\n", []catalogue.Item{{ID: 1, Name: "Bolt", Stock: transferStock(0)}})
	transferRejectImport(t, "id,name,stock\n1,Bolt,\n0001,Washer,0\n")
}

type transferWatchedReader struct {
	*strings.Reader
	readBytes int
	closed    bool
}

func (r *transferWatchedReader) Read(p []byte) (int, error) {
	n, err := r.Reader.Read(p)
	r.readBytes += n
	return n, err
}

func (r *transferWatchedReader) Close() error { r.closed = true; return nil }

func TestTransferCSVRawByteBudget(t *testing.T) {
	base := "id,name,stock\n1,Bolt,0\n"
	exact := base + strings.Repeat("\n", catalogue.MaxBytes-len(base))
	for _, extra := range []int{0, 1, 4096} {
		t.Run(strconv.Itoa(extra), func(t *testing.T) {
			r := &transferWatchedReader{Reader: strings.NewReader(exact + strings.Repeat("\n", extra))}
			calls := 0
			err := catalogue.Import(r, func(items []catalogue.Item) error {
				calls++
				if len(items) != 1 || items[0].ID != 1 {
					t.Errorf("replacement = %#v", items)
				}
				return nil
			})
			if extra == 0 && (err != nil || calls != 1 || r.readBytes != catalogue.MaxBytes) {
				t.Errorf("exact limit: error = %v, calls = %d, bytes = %d", err, calls, r.readBytes)
			}
			if extra > 0 && (err == nil || calls != 0) {
				t.Errorf("oversize: error = %v, calls = %d", err, calls)
			}
			if r.readBytes > catalogue.MaxBytes+1 || r.closed {
				t.Errorf("reader ownership/budget: bytes = %d, closed = %v", r.readBytes, r.closed)
			}
		})
	}
}

func TestTransferCSVFieldAndRowLimits(t *testing.T) {
	name := strings.Repeat("n", catalogue.MaxFieldBytes)
	transferImportItems(t, "id,name,stock\n1,"+name+",\n", []catalogue.Item{{ID: 1, Name: name}})
	quotedName := strings.Repeat("\"", catalogue.MaxFieldBytes)
	transferImportItems(t, "id,name,stock\n1,\""+strings.Repeat("\"\"", catalogue.MaxFieldBytes)+"\",\n", []catalogue.Item{{ID: 1, Name: quotedName}})
	transferRejectImport(t, "id,name,stock\n1,Bolt,\n2,"+name+"n,\n")
	var input strings.Builder
	input.WriteString("id,name,stock\n")
	want := make([]catalogue.Item, 0, catalogue.MaxRows)
	for i := 1; i <= catalogue.MaxRows; i++ {
		id := uint64(i)
		input.WriteString(strconv.FormatUint(id, 10) + ",Bolt,\n")
		want = append(want, catalogue.Item{ID: id, Name: "Bolt"})
	}
	transferImportItems(t, input.String(), want)
	transferRejectImport(t, input.String()+strconv.Itoa(catalogue.MaxRows+1)+",Washer,\n")
}

type transferTerminalFailureReader struct {
	*strings.Reader
	cause error
}

func (r *transferTerminalFailureReader) Read(p []byte) (int, error) {
	if r.Len() == 0 {
		return 0, r.cause
	}
	return r.Reader.Read(p)
}

func TestTransferCSVReaderFailureBeforeReplacement(t *testing.T) {
	cause := errors.New("upload connection failed")
	r := &transferTerminalFailureReader{Reader: strings.NewReader("id,name,stock\n1,Bolt,2\n2,Washer,3\n"), cause: cause}
	calls := 0
	err := catalogue.Import(r, func([]catalogue.Item) error { calls++; return nil })
	if !errors.Is(err, cause) || calls != 0 {
		t.Fatalf("Import error = %v, replacement calls = %d; want reader cause before replacement", err, calls)
	}
}

func TestTransferCSVCallbackErrorAndBorrowedReader(t *testing.T) {
	cause := errors.New("catalogue storage failed")
	r := &transferWatchedReader{Reader: strings.NewReader("id,name,stock\n1,Bolt,0\n")}
	calls := 0
	err := catalogue.Import(r, func([]catalogue.Item) error { calls++; return cause })
	if !errors.Is(err, cause) || calls != 1 || r.closed {
		t.Fatalf("Import error = %v, replacement calls = %d, reader closed = %v", err, calls, r.closed)
	}
}
