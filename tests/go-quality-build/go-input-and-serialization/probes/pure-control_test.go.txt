package tally_test

import (
	"testing"

	tally "example.com/oddtally"
)

func TestNegativeOddValuesContribute(t *testing.T) {
	for _, tc := range []struct {
		values []int
		want   int
	}{
		{[]int{-7, -4, -3, 0, 2, 5}, -5},
		{[]int{-1}, -1},
		{nil, 0},
		{[]int{2, 4, -6}, 0},
	} {
		if got := tally.Total(tc.values); got != tc.want {
			t.Errorf("Total(%v) = %d, want %d", tc.values, got, tc.want)
		}
	}
}
