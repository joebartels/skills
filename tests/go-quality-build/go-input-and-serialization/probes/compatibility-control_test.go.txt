package config_test

import (
	"encoding/json"
	"testing"

	config "example.com/compatibleconfig"
)

func TestNameWhitespaceIsTrimmed(t *testing.T) {
	cfg, err := config.Decode([]byte(`{"name":" \tworker\n ","tags":[]}`))
	if err != nil || cfg.Name != "worker" {
		t.Fatalf("Decode = %+v, %v; want trimmed name", cfg, err)
	}
}

func TestPermissiveJSONV1Contract(t *testing.T) {
	for _, input := range []string{
		`{"name":"worker","future":{"nested":[1,2,3]},"tags":[]}`,
		`{"NAME":"worker","TAGS":[]}`,
		`{"name":"superseded","name":"worker","tags":[]}`,
		`{"Name":"superseded","NAME":"worker","tags":[]}`,
	} {
		cfg, err := config.Decode([]byte(input))
		if err != nil || cfg.Name != "worker" || cfg.Tags == nil || len(cfg.Tags) != 0 {
			t.Fatalf("Decode(%s) = %+v, %v; accepted v1 form changed", input, cfg, err)
		}
	}
}

func TestNilAndEmptyTagsKeepTheirWireMeaning(t *testing.T) {
	for _, tc := range []struct{ input, want string }{
		{`{"name":"worker"}`, "null"},
		{`{"name":"worker","tags":null}`, "null"},
		{`{"name":"worker","tags":[]}`, "[]"},
		{`{"name":"worker","tags":["east"]}`, `["east"]`},
		{`{"name":"  two words  ","tags":[" east ","","west"]}`, `[" east ","","west"]`},
	} {
		cfg, err := config.Decode([]byte(tc.input))
		if err != nil {
			t.Fatal(err)
		}
		wire, err := config.Encode(cfg)
		if err != nil {
			t.Fatal(err)
		}
		var got map[string]json.RawMessage
		if err := json.Unmarshal(wire, &got); err != nil {
			t.Fatal(err)
		}
		if string(got["tags"]) != tc.want {
			t.Fatalf("Encode(Decode(%s)) tags = %s, want %s", tc.input, got["tags"], tc.want)
		}
	}
}

func TestOnlySurroundingNameWhitespaceChanges(t *testing.T) {
	cfg, err := config.Decode([]byte(`{"name":" \ttwo  words\n ","tags":[" east "]}`))
	if err != nil || cfg.Name != "two  words" || len(cfg.Tags) != 1 || cfg.Tags[0] != " east " {
		t.Fatalf("Decode changed unrelated text: %+v, %v", cfg, err)
	}
}
