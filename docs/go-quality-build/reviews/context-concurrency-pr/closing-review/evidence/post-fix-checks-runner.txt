GATE = 'tests/go-quality-build/results/2026-10-02-context-concurrency-delivery/verify-delivery.py'

def save():
    (OUT / 'post-fix-checks.json').write_text(json.dumps(report, indent=2) + '\n')

def run(label, argv, cwd=ROOT, env=None, timeout=60):
    process_env = os.environ.copy()
    process_env.pop('PYTHONOPTIMIZE', None)
    process_env.update(env or {})
    start = time.monotonic()
    proc = subprocess.run(argv, cwd=cwd, env=process_env, capture_output=True, text=True, timeout=timeout)
    row = {'id': label, 'argv': argv, 'cwd': str(cwd), 'environment_removed': ['PYTHONOPTIMIZE'],
           'environment_overrides': env or {}, 'exit_code': proc.returncode,
           'elapsed_seconds': round(time.monotonic()-start, 3), 'stdout': proc.stdout, 'stderr': proc.stderr}
    report['checks'].append(row)
    save()
    print(label, proc.returncode, proc.stdout.strip()[:150], proc.stderr.strip()[-300:], flush=True)
    return proc

sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
tree = lambda p: {x.relative_to(p).as_posix(): sha(x) for x in p.rglob('*') if x.is_file()}

protected = [
    'docs/go-quality-build/reviews/concurrency-readiness/preflight/example',
    'docs/go-quality-build/reviews/concurrency-readiness/preflight/raw/verify_example.py',
    'docs/go-quality-build/reviews/concurrency-readiness/raw/verify-readiness.py',

    'plugins/go-quality-build', 'plugins/go-quality-review',

    'tests/go-quality-build/results/2026-10-02-context-concurrency-combined',
    ORIGINAL,
]
comparison = []
for rev in [BASE, HEAD]:
    argv = ['rtk', 'proxy', 'git', 'ls-tree', '-r', rev, '--', *protected]
    p = subprocess.run(argv, cwd=ROOT, capture_output=True)
    assert p.returncode == 0, p.stderr
    comparison.append(p.stdout)
    report.setdefault('protected_identity_commands', []).append({'argv': argv, 'exit_code': p.returncode,
        'tracked_entries': len(p.stdout.splitlines()), 'tree_inventory_sha256': hashlib.sha256(p.stdout).hexdigest()})
assert comparison[0] == comparison[1]
report['protected_source_tree_identity_equal'] = True
save()

        assert hashlib.sha256(old.stdout).hexdigest() == value
        report['revision2_snapshot_matches_c5_delivery'].append({'path': name, 'sha256': value,
